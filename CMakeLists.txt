cmake_minimum_required(VERSION 3.25)
project(FruityOriginalEngine VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Windows-only VST3 and x86 worker.")
endif()
include(CTest)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
 add_executable(fruity_probe32 src/legacy/probe32.cpp)
 target_compile_features(fruity_probe32 PRIVATE cxx_std_23)
 set_target_properties(fruity_probe32 PROPERTIES OUTPUT_NAME "125A-Fruity-Probe32")
 add_executable(fruity_helper32 src/helper/Worker32.cpp src/bridge/Protocol.hpp)
 target_include_directories(fruity_helper32 PRIVATE src)
 target_compile_features(fruity_helper32 PRIVATE cxx_std_23)
 set_target_properties(fruity_helper32 PROPERTIES OUTPUT_NAME "125A-Fruity-Helper32")
 if(MSVC)
  target_compile_options(fruity_probe32 PRIVATE /W4 /WX)
  target_compile_options(fruity_helper32 PRIVATE /W4 /WX)
 endif()
 if(BUILD_TESTING)
  add_library(fruity_mock_engine SHARED tests/mock_engine.cpp tests/mock_engine.def)
  target_compile_features(fruity_mock_engine PRIVATE cxx_std_23)
  if(MSVC)
   target_compile_options(fruity_mock_engine PRIVATE /W4 /WX)
  endif()
  add_test(NAME x86_mock_pcm_contract
   COMMAND $<TARGET_FILE:fruity_probe32> $<TARGET_FILE:fruity_mock_engine>
   "${CMAKE_CURRENT_SOURCE_DIR}/tests/mock-bank.txt"
   "${CMAKE_CURRENT_BINARY_DIR}/mock-output.wav")
  add_test(NAME x86_missing_asset_is_rejected
   COMMAND $<TARGET_FILE:fruity_probe32> $<TARGET_FILE:fruity_mock_engine>
   "${CMAKE_CURRENT_SOURCE_DIR}/tests/DOES-NOT-EXIST.sf2"
   "${CMAKE_CURRENT_BINARY_DIR}/should-not-exist.wav")
  set_tests_properties(x86_missing_asset_is_rejected PROPERTIES WILL_FAIL TRUE)
 endif()
elseif(CMAKE_SIZEOF_VOID_P EQUAL 8)
 if(NOT DEFINED VST3_SDK_ROOT OR NOT EXISTS "${VST3_SDK_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "VST3_SDK_ROOT must name a pinned recursive VST3 SDK checkout")
 endif()
 set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_BUNDLE_FOR_WINDOWS OFF CACHE BOOL "" FORCE)
 set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
 add_subdirectory("${VST3_SDK_ROOT}" "${CMAKE_BINARY_DIR}/steinberg")
 file(TO_CMAKE_PATH "${VST3_SDK_ROOT}" _sdk)
 smtg_add_vst3plugin(FruitySoundFont
   "${_sdk}/public.sdk/source/main/dllmain.cpp"
   src/vst3/WorkerBridge.cpp
   src/vst3/Processor.cpp
   src/vst3/Controller.hpp
   src/vst3/Factory.cpp)
 target_compile_features(FruitySoundFont PRIVATE cxx_std_23)
 target_include_directories(FruitySoundFont PRIVATE src)
 target_compile_definitions(FruitySoundFont PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(FruitySoundFont PRIVATE sdk)
 set_target_properties(FruitySoundFont PROPERTIES
   OUTPUT_NAME "125A Fruity SoundFont Player")
 if(MSVC)
  target_compile_options(FruitySoundFont PRIVATE /W4)
 endif()
else()
 message(FATAL_ERROR "Unsupported target architecture")
endif()
